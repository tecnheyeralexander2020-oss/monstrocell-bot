function getReply(text){
  text = text.toLowerCase();
  if(text.includes('iphone') || text.includes('apple')) return CATALOG.iphone;
  if(text.includes('samsung') || text.includes('a05') || text.includes('a07')) return CATALOG.samsung;
  if(text.includes('tv') || text.includes('televisor') || text.includes('iffalcon') || text.includes('samsung tv')) return CATALOG.tv;
  if(text.includes('precio') || text.includes('credito') || text.includes('cuota') || text.includes('inicial')) return CATALOG.general + "\n\n" + CATALOG.iphone;
  return CATALOG.general;
}

async function sendMessage(to, message){
  const url = `https://graph.facebook.com/v20.0/${PHONE_NUMBER_ID}/messages`;
  await axios.post(url, {
    messaging_product: 'whatsapp',
    to: to,
    type: 'text',
    text: { body: message }
  }, {
    headers: { Authorization: `Bearer ${ACCESS_TOKEN}` }
  });
}

// Verificación webhook
app.get('/webhook', (req,res)=>{
  const mode = req.query['hub.mode'];
  const token = req.query['hub.verify_token'];
  const challenge = req.query['hub.challenge'];
  if(mode === 'subscribe' && token === VERIFY_TOKEN){
    console.log('Webhook verificado');
    res.status(200).send(challenge);
  } else {
    res.sendStatus(403);
  }
});

// Recibir mensajes
app.post('/webhook', async (req,res)=>{
  try{
    const entry = req.body.entry?.[0];
    const change = entry?.changes?.[0];
    const value = change?.value;
    const msg = value?.messages?.[0];
    if(msg && msg.type === 'text'){
      const from = msg.from;
      const text = msg.text.body;
      console.log(`Mensaje de ${from}: ${text}`);
      const reply = getReply(text);
      // Delay 5-10s para parecer humano
      setTimeout(async ()=>{
        await sendMessage(from, reply);
        console.log('Respondido a', from);
      }, 7000);
    }
    res.sendStatus(200);
  } catch(e){
    console.error(e);
    res.sendStatus(200);
  }
});

const PORT = process.env.PORT || 3000;
